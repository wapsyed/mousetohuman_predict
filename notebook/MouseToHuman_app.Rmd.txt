# Loaded lazily on the first prediction so the app starts faster.
models <- local({
  cache <- NULL
  function() {
    if (is.null(cache)) {
      cache <<- list(
        shared    = readRDS(here("Modelling", "Models", "rf_model_shared.rds")),
        rank      = readRDS(here("Modelling", "Models", "rf_model_rank.rds")),
        direction = readRDS(here("Modelling", "Models", "rf_model_direction.rds"))
      )
    }
    cache
  }
})

# --- Annotation tables (biological layers; always BTM-based for the model) --
alignments_human2mouse_all <- readRDS(here("tables", "alignments_human2mouse_all.rds"))
human_mouse_cds_distance <- readRDS(here("tables", "human_mouse_cds_distance.rds"))
gene_tf_shared_unique <- readRDS(here("tables", "gene_tf_shared_unique.rds"))

# Spearman rho + BH-adjusted p annotation
cor_label <- function(df, x, y) {
  tibble(rho = cor(df[[x]], df[[y]], method = "spearman", use = "complete.obs"),
         p = cor.test(df[[x]], df[[y]], method = "spearman", exact = FALSE)$p.value) %>%
    mutate(padj = p.adjust(p, method = "BH"),
           label = sprintf("rho=%.2f\npadj=%.1e", rho, padj))
}

# GSEA helper (uses the chosen gene set). clusterProfiler is loaded lazily on
# the first run to keep app startup fast.
run_gsea <- function(dge, geneset) {
  suppressPackageStartupMessages(library(clusterProfiler))
  term2gene <- geneset %>%
    select(process, genes = hgnc_symbol)
  df2 <- dge %>%
    distinct() %>%
    select(condition, genes = hgnc_symbol, log2fold_change = mean_l2fc, adj_p_val) %>%
    filter(!is.na(genes), !is.na(log2fold_change)) %>%
    mutate(rank = log2fold_change * -log(adj_p_val)) %>%
    select(condition, genes, rank)
  gsea_output <- autoGSEA(df = df2, TERM2GENE = term2gene, geneset_name = "user")
  gsea_output$gsea %>%
    as.data.frame() %>%
    clean_names() %>%
    rename(process = id) %>%
    filter(!is.na(process)) %>%
    rownames_to_column("del") %>%
    select(-del, -description) %>%
    rename(p_adjust_gsea = p_adjust,
           q_value_gsea = qvalue,
           rank_gsea = rank)
}

# Prediction helper
predict_scores <- function(d) {
  m <- models()
  d %>%
    bind_cols(predict(m$rank, new_data = d, type = "numeric") %>% rename(score_rank = .pred)) %>%
    bind_cols(predict(m$direction, new_data = d, type = "prob") %>% select(score_direction = .pred_TRUE)) %>%
    bind_cols(predict(m$shared, new_data = d, type = "prob") %>% select(score_shared = .pred_TRUE)) %>%
    mutate(score_translational = score_rank * score_direction)
}

# --- Feature heatmap helpers --------------------------------------------------
# Features shown in the heatmaps (readable label = column name)
heat_map <- c(
  "Log2FC (mouse)"        = "mean_log2fc_mouse",
  "Absolute rank (mouse)" = "rank_mouse",
  "1/SE"                  = "inverse_se_mouse",
  "Protein identity"      = "identity_human2mouse",
  "Kimura"                = "dist_k80",
  "nTotal TFs"            = "n_tf_total",
  "%Shared TFs"           = "pct_tf_shared",
  "nTotal cCREs"          = "n_total_cres_gene",
  "%Match Type (PLS)"     = "pct_match_type_PLS",
  "%Match Type (pELS)"    = "pct_match_type_pELS",
  "%Match Type (dELS)"    = "pct_match_type_dELS",
  "CTCF match"            = "pct_match_ctcf_dELS_CTCF-bound"
)

build_heat_mat <- function(df) {
  df %>%
    distinct(hgnc_symbol, .keep_all = TRUE) %>%
    select(hgnc_symbol, all_of(unname(heat_map))) %>%
    setNames(c("hgnc_symbol", names(heat_map))) %>%
    column_to_rownames("hgnc_symbol") %>%
    as.matrix()
}

# Rows = genes, columns = features. z-scores are computed relative to `baseline`
# (all genes) so the colours are comparable across different gene selections.
feature_heatmap <- function(genes_df, baseline_mat, title) {
  base_mean <- colMeans(baseline_mat, na.rm = TRUE)
  base_sd   <- apply(baseline_mat, 2, sd, na.rm = TRUE)
  heat_dat  <- build_heat_mat(genes_df)
  heat_z    <- sweep(sweep(heat_dat, 2, base_mean, "-"), 2, base_sd, "/")
  heat_z[is.na(heat_z)] <- 0
  gene_levels <- rownames(heat_z)[hclust(dist(heat_z))$order]
  feat_levels <- colnames(heat_z)[hclust(dist(t(heat_z)))$order]
  heat_dat %>%
    as.data.frame() %>%
    rownames_to_column("hgnc_symbol") %>%
    pivot_longer(-hgnc_symbol, names_to = "feature", values_to = "value") %>%
    mutate(z = (value - base_mean[feature]) / base_sd[feature],
           hgnc_symbol = factor(hgnc_symbol, levels = gene_levels),
           feature = factor(feature, levels = feat_levels)) %>%
    ggplot() +
    aes(x = feature, y = hgnc_symbol, fill = z,
        text = paste0(hgnc_symbol, "\n", feature, ": ", round(value, 2))) +
    geom_tile(color = "white", linewidth = 0.3) +
    scale_fill_gradientn(colours = c("#4DBBD5FF", "white", "#4361ee"),
                         limits = c(-2.5, 2.5), oob = scales::squish) +
    theme_vaxgo() +
    theme(axis.text.x = element_text(angle = 45, hjust = 1, size = 7),
          axis.text.y = element_text(size = 7)) +
    labs(x = "", y = "", fill = "z-score (all genes)", title = title)
}

# --- Score heatmap helper -----------------------------------------------------
# Rows = genes, columns = the four model scores (z-scored across ALL genes).
score_heat_cols <- c(
  "Shared"        = "score_shared",
  "Rank"          = "score_rank",
  "Direction"     = "score_direction",
  "Translational" = "score_translational"
)

score_heatmap <- function(genes_df, baseline_df, title) {
  score_mat <- function(df) {
    df %>%
      distinct(hgnc_symbol, .keep_all = TRUE) %>%
      select(hgnc_symbol, all_of(unname(score_heat_cols))) %>%
      setNames(c("hgnc_symbol", names(score_heat_cols))) %>%
      column_to_rownames("hgnc_symbol") %>%
      as.matrix()
  }
  base_mat  <- score_mat(baseline_df)
  base_mean <- colMeans(base_mat, na.rm = TRUE)
  base_sd   <- apply(base_mat, 2, sd, na.rm = TRUE)
  heat_dat  <- score_mat(genes_df)
  heat_z    <- sweep(sweep(heat_dat, 2, base_mean, "-"), 2, base_sd, "/")
  heat_z[is.na(heat_z)] <- 0
  gene_levels <- rownames(heat_z)[hclust(dist(heat_z))$order]
  heat_dat %>%
    as.data.frame() %>%
    rownames_to_column("hgnc_symbol") %>%
    pivot_longer(-hgnc_symbol, names_to = "score", values_to = "value") %>%
    mutate(z = (value - base_mean[score]) / base_sd[score],
           hgnc_symbol = factor(hgnc_symbol, levels = gene_levels),
           score = factor(score, levels = colnames(heat_dat))) %>%
    ggplot() +
    aes(x = score, y = hgnc_symbol, fill = z,
        text = paste0(hgnc_symbol, "\n", score, ": ", round(value, 3))) +
    geom_tile(color = "white", linewidth = 0.3) +
    scale_fill_gradientn(colours = c("#4DBBD5FF", "white", "#4361ee"),
                         limits = c(-2.5, 2.5), oob = scales::squish) +
    theme_vaxgo() +
    theme(axis.text.x = element_text(angle = 0, hjust = 0.5, size = 8),
          axis.text.y = element_text(size = 7)) +
    labs(x = "", y = "", fill = "z-score (all genes)", title = title)
}

# --- Input table ---
dge_data <- reactive({
  if (input$dge_source == "example") {
    readRDS(here("Run your analysis here", "example_dge_result.rds"))
  } else {
    req(input$dge_file)
    ext <- tools::file_ext(input$dge_file$name)
    if (ext == "rds") {
      readRDS(input$dge_file$datapath)
    } else {
      readr::read_csv(input$dge_file$datapath, show_col_types = FALSE)
    }
  }
})

# --- Gene set for GSEA (BTM or custom) ---
geneset_data <- reactive({
  if (input$geneset_source == "BTM (default)") {
    btm_annotation_genes
  } else {
    req(input$geneset_file)
    ext <- tools::file_ext(input$geneset_file$name)
    gs <- if (ext == "rds") {
      readRDS(input$geneset_file$datapath)
    } else {
      readr::read_csv(input$geneset_file$datapath, show_col_types = FALSE)
    }
    gs %>%
      rename(process = 1, hgnc_symbol = 2) %>%
      distinct()
  }
})

# --- Annotate features (always BTM-based for the model) ---
annotated_data <- reactive({
  dge <- dge_data()

  btm_wide <- bind_rows(
    btm_annotation_genes %>% select(hgnc_symbol, group),
    dge %>% select(hgnc_symbol) %>%
      anti_join(btm_annotation_genes %>% select(hgnc_symbol) %>% distinct(), by = "hgnc_symbol") %>%
      mutate(group = "Not-BTM")
  ) %>%
    group_by(hgnc_symbol, group) %>%
    summarise(n = n(), .groups = "drop") %>%
    pivot_wider(names_from = group, values_from = n) %>%
    column_to_rownames("hgnc_symbol") %>%
    as.matrix() %>%
    replace(is.na(.), 0) %>%
    as.data.frame() %>%
    rownames_to_column("hgnc_symbol")

  layers <- human_mouse_cds_distance %>%
    select(hgnc_symbol, dist_k80) %>%
    inner_join(alignments_human2mouse_all, by = join_by("hgnc_symbol" == "gene")) %>%
    inner_join(btm_wide, by = "hgnc_symbol") %>%
    inner_join(gene_tf_shared_unique, by = join_by("hgnc_symbol" == "gene")) %>%
    inner_join(cres_type_homology_comparison_wide %>%
                 select(hgnc_symbol, 'n_total_cres_gene':'pct_match_ctcf_PLS_CTCF-bound'),
               by = "hgnc_symbol")

  dge %>%
    rename(mean_log2fc_mouse = mean_l2fc,
           inverse_se_mouse = inverse_se,
           p_adj_mouse = adj_p_val,
           timepoint_comparison = timepoint) %>%
    mutate(sign_mouse = sign(mean_log2fc_mouse)) %>%
    group_by(treatment, pathogen, timepoint_comparison) %>%
    mutate(rank_mouse = percent_rank(abs(mean_log2fc_mouse) * -log10(pmax(p_adj_mouse, 1e-300))) * 100) %>%
    ungroup() %>%
    inner_join(layers, by = "hgnc_symbol") %>%
    filter(hgnc_symbol %in% btm_annotation_genes$hgnc_symbol) %>%
    group_by(treatment, pathogen, timepoint_comparison) %>%
    mutate(rank_mouse_btm = percent_rank(abs(mean_log2fc_mouse) * -log10(pmax(p_adj_mouse, 1e-300))) * 100) %>%
    ungroup()
})

# --- Button-gated pipeline with progress ---
results <- eventReactive(input$run, {
  withProgress(message = "Running analysis...", value = 0, {
    incProgress(0.15, detail = "Reading input")
    d <- annotated_data()

    incProgress(0.25, detail = "Running GSEA")
    g <- run_gsea(dge_data(), geneset_data())

    incProgress(0.35, detail = "Predicting scores")
    p <- predict_scores(d)

    incProgress(0.25, detail = "Done")
    list(predictions = p, gsea = g)
  })
})

# --- Apply the user-defined cutoffs ---
selected <- reactive({
  req(results())
  results()$predictions %>%
    filter(score_translational >= input$cut_translational,
           score_shared >= input$cut_shared,
           score_rank >= input$cut_rank,
           score_direction >= input$cut_direction)
})

# --- Enriched GSEA processes passing the padj cutoff ---
enriched_processes <- reactive({
  req(results())
  results()$gsea %>%
    filter(p_adjust_gsea < input$padj_filter) %>%
    distinct(process) %>%
    arrange(process) %>%
    pull(process)
})

# --- Mouse leading-edge genes of the selected process ---
process_genes <- reactive({
  req(results(), input$process_select)
  results()$gsea %>%
    filter(process == input$process_select) %>%
    pull(core_enrichment) %>%
    strsplit("/") %>%
    unlist() %>%
    unique()
})

# --- Keep the sidebar process dropdown in sync with the results ------------
if (exists("session", inherits = TRUE)) {
  observe({
    updateSelectInput(session, "process_select", choices = enriched_processes())
  })
}
